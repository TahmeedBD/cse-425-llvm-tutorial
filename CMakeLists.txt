cmake_minimum_required(VERSION 3.20)
project(KaleidoscopeTutorial CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(toy_lexer src/lexer.cpp)
target_include_directories(toy_lexer PUBLIC src)
add_executable(toy toy.cpp)
target_link_libraries(toy PRIVATE toy_lexer)

# Enable in later chapters when LLVM IR generation is implemented.
option(TOY_WITH_LLVM "Link LLVM libraries" OFF)
if(TOY_WITH_LLVM)
    find_package(LLVM REQUIRED CONFIG)
    message(STATUS "Found LLVM ${LLVM_PACKAGE_VERSION}")
    target_include_directories(toy PRIVATE ${LLVM_INCLUDE_DIRS})
    separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
    target_compile_options(toy PRIVATE ${LLVM_DEFINITIONS_LIST})
    llvm_map_components_to_libnames(llvm_libs core orcjit native)
    target_link_libraries(toy PRIVATE ${llvm_libs})
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(lexer_test tests/lexer_test.cpp)
    target_link_libraries(lexer_test PRIVATE toy_lexer)
    add_test(NAME lexer COMMAND lexer_test)
endif()
